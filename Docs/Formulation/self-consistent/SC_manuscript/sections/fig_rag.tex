% The reaction admissibility graph for the nine families. Drawn here rather
% than lifted from the mean-field treatment: that graph is a size-resolved
% ladder over four families, and this model carries moments over nine, with a
% basal chain and a grain-boundary loop channel that the ladder has no vertex
% or edge for.
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
  font=\small, >=Latex,
  mob/.style={draw, rounded corners=2pt, minimum height=7mm, minimum width=11mm,

  ipop/.style={draw, rounded corners=2pt, align=center, fill=red!8,
               draw=red!60, minimum height=9mm, inner sep=3pt},
  vpop/.style={draw, rounded corners=2pt, align=center, fill=cyan!10,
               draw=cyan!55!black, minimum height=9mm, inner sep=3pt},
  bpop/.style={draw, rounded corners=2pt, align=center, fill=violet!10,
               draw=violet!60, minimum height=9mm, inner sep=3pt},
  ext/.style={draw, align=center, fill=black!6, draw=black!45,
              minimum height=8mm, inner sep=3pt},
  capture/.style={->, thick, blue!65},
  emitE/.style={->, dashed, orange!85!black},
  rec/.style={<->, very thick, red!75},

  gbl/.style={->, very thick, green!45!black}]

  \node[ext] (G) at (0,4.6) {cascade source $G$~(P6)};

  % ---- positive polarity ------------------------------------------------
  \node[mob] (Ci)  at (-4.8,2.4) {$C_i$};
  \node[mob] (C2i) at (-3.1,3.5) {$C_{2i}$};
  \node[mob] (C3i) at (-1.6,3.5) {$C_{3i}$};
  \node[ipop] (ai) at (3.2,2.4) {$(i,a_1),(i,a_2),(i,a_3)$ \\
                                 \scriptsize prismatic $\langle a\rangle$,
                                 interstitial};
  \node[anchor=west, font=\itshape, text=black!60] at (-6.6,3.9)
       {positive polarity};

  % ---- datum ------------------------------------------------------------
  \draw[ultra thick] (-6.6,0) -- (7.8,0);
  \node[anchor=east, text=black!55, font=\scriptsize] at (7.8,0.3)

  % ---- negative polarity ------------------------------------------------
  \node[mob] (Cv) at (-4.8,-2.4) {$C_v$};
  \node[vpop] (av) at (0.6,-1.4) {$(v,a_1),(v,a_2),(v,a_3)$ \\

\caption{Reaction admissibility graph for the nine families of
\cref{eq:families}. Vertices above the datum carry positive (interstitial)
polarity, those below it negative (vacancy) polarity. Blue arrows are monomer
capture (P2), which acts on a vacancy loop with the opposite sign when the
captured species is an interstitial; dashed orange is thermal emission (P5);
the heavy red double arrow is recombination (P1); orange loops are coalescence
(P3); violet is the one-way basal chain $c_0\to c_f\to c_p$ (P7) and the
loop--network transfer (P8); gray is absorption at the network (P4) and the
Dirichlet condition on the mobile species at a grain boundary (P9). The green
edges (P10) are the channel introduced in \cref{sec:gb-loops}, by which a loop
that has grown large enough to intersect a boundary is absorbed by it --- an
edge with no counterpart in a mean-field treatment, where a loop has no size
relative to a boundary that is itself only a sink strength. The three prismatic
variants of each character are drawn as one vertex for legibility; they are
separate populations that exchange nothing with one another and differ only in
habit normal.}
\label{fig:rag}
\end{figure}
